"use client";

import { useCallback, useSyncExternalStore } from "react";
import dayjs from "dayjs";
import relativeTime from "dayjs/plugin/relativeTime";

import { getNextWidgetTickDelay } from "./use-widget-now";

dayjs.extend(relativeTime);

const listeners = new Set<() => void>();
let timeout: ReturnType<typeof setTimeout> | undefined;

const tick = () => {
  timeout = setTimeout(tick, getNextWidgetTickDelay(new Date(), "minute"));
  for (const listener of listeners) listener();
};

const handleVisibilityChange = () => {
  if (document.visibilityState !== "visible") return;
  clearTimeout(timeout);
  tick();
};

const subscribeToMinute = (listener: () => void) => {
  listeners.add(listener);
  if (listeners.size === 1) {
    tick();
    document.addEventListener("visibilitychange", handleVisibilityChange);
  }
  return () => {
    listeners.delete(listener);
    if (listeners.size !== 0) return;
    clearTimeout(timeout);
    document.removeEventListener("visibilitychange", handleVisibilityChange);
  };
};

export const useRowMinuteTime = (value: string | number | null, short: boolean) => {
  const getSnapshot = useCallback(() => {
    if (value === null) return "-";
    return dayjs(value).fromNow(short);
  }, [value, short]);
  return useSyncExternalStore(subscribeToMinute, getSnapshot, getSnapshot);
};
